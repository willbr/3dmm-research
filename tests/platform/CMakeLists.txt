cmake_minimum_required(VERSION 3.22)
project(KauaiPlatformChecks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
enable_testing()
find_package(Threads REQUIRED)
set(platform_dir "${CMAKE_CURRENT_LIST_DIR}/../../kauai/src")
add_library(platform-services STATIC "${platform_dir}/core_platform.cpp" "${platform_dir}/platform_text.cpp")
target_sources(platform-services PRIVATE "${platform_dir}/user_paths.cpp")
target_include_directories(platform-services PUBLIC "${platform_dir}")
target_link_libraries(platform-services PUBLIC Threads::Threads)
if(WIN32)
  target_sources(platform-services PRIVATE "${platform_dir}/platform_files_win.cpp")
  target_sources(platform-services PRIVATE "${platform_dir}/platform_users_win.cpp")
  target_link_libraries(platform-services PUBLIC mpr shell32)
else()
  target_sources(platform-services PRIVATE "${platform_dir}/platform_files_posix.cpp")
  target_sources(platform-services PRIVATE "${platform_dir}/platform_users_posix.cpp")
endif()
foreach(check IN ITEMS platform_services file_services user_services text_services)
  add_executable(${check}_test "../${check}_test.cpp")
  target_link_libraries(${check}_test PRIVATE platform-services)
  add_test(NAME ${check} COMMAND ${check}_test)
endforeach()
add_executable(text_portable_test ../text_services_test.cpp "${platform_dir}/platform_text.cpp")
target_include_directories(text_portable_test PRIVATE "${platform_dir}")
target_compile_definitions(text_portable_test PRIVATE KAUAI_PORTABLE_TEXT)
target_link_libraries(text_portable_test PRIVATE Threads::Threads)
add_test(NAME text_portable COMMAND text_portable_test)
