cmake_minimum_required(VERSION 3.22)
project(KauaiPlatformChecks LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
enable_testing()
find_package(Threads REQUIRED)
set(platform_dir "${CMAKE_CURRENT_LIST_DIR}/../../kauai/src")
set(engine_dir "${CMAKE_CURRENT_LIST_DIR}/../../libraries/engine")
add_library(platform-engine-text STATIC "${engine_dir}/src/text.c" "${engine_dir}/src/heap.c")
target_sources(platform-engine-text PRIVATE "${engine_dir}/src/settings.c" "${engine_dir}/src/settings_paths.c")
target_include_directories(platform-engine-text PUBLIC "${engine_dir}/include")
set_target_properties(platform-engine-text PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
add_library(platform-services STATIC "${platform_dir}/core_platform.cpp" "${platform_dir}/platform_text.cpp" "${platform_dir}/text_utf8.cpp")
target_sources(platform-services PRIVATE "${platform_dir}/platform_users_bridge.cpp")
target_sources(platform-services PRIVATE "${platform_dir}/settings_store_bridge.cpp")
target_include_directories(platform-services PUBLIC "${platform_dir}")
target_link_libraries(platform-services PUBLIC Threads::Threads platform-engine-text)
add_library(platform-native-files STATIC)
target_include_directories(platform-native-files PUBLIC "${engine_dir}/include")
set_target_properties(platform-native-files PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
target_sources(platform-services PRIVATE "${platform_dir}/platform_files_bridge.cpp")
add_library(platform-native-users STATIC "${engine_dir}/src/native_user_paths.c")
target_include_directories(platform-native-users PUBLIC "${engine_dir}/include")
set_target_properties(platform-native-users PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
target_link_libraries(platform-services PUBLIC platform-native-files platform-native-users)
add_library(platform-native-settings STATIC)
target_include_directories(platform-native-settings PUBLIC "${engine_dir}/include")
set_target_properties(platform-native-settings PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
target_link_libraries(platform-native-settings PUBLIC platform-engine-text)
target_link_libraries(platform-services PUBLIC platform-native-settings)
if(WIN32)
  target_sources(platform-native-files PRIVATE "${engine_dir}/src/native_files_win.c")
  target_link_libraries(platform-native-files PUBLIC user32 shell32)
  target_sources(platform-native-users PRIVATE "${engine_dir}/src/native_users_win.c")
  target_link_libraries(platform-native-users PUBLIC mpr shell32)
  target_sources(platform-native-settings PRIVATE "${engine_dir}/src/native_settings_win.c")
  target_link_libraries(platform-native-settings PUBLIC advapi32)
  set(settings_native_test settings_store_win)
else()
  target_sources(platform-native-files PRIVATE "${engine_dir}/src/native_files_posix.c")
  target_sources(platform-native-users PRIVATE "${engine_dir}/src/native_users_posix.c")
  target_sources(platform-native-settings PRIVATE "${engine_dir}/src/native_settings_posix.c")
  target_link_libraries(platform-native-settings PUBLIC platform-native-files Threads::Threads)
  set(settings_native_test settings_store_posix)
endif()
if(UNIX AND NOT APPLE)
  add_executable(native_files_posix_runtime_test ../native_files_posix_runtime_test.cpp)
  target_link_libraries(native_files_posix_runtime_test PRIVATE platform-services)
  # GNU linker interposition exercises EINTR and short transfers on Linux.
  target_link_options(native_files_posix_runtime_test PRIVATE
    "-Wl,--wrap=read" "-Wl,--wrap=write" "-Wl,--wrap=fsync" "-Wl,--wrap=open" "-Wl,--wrap=flock")
  add_test(NAME native_files_posix_runtime COMMAND native_files_posix_runtime_test)
endif()
if(NOT WIN32)
  add_executable(native_files_c_test ../native_files_c_test.c)
  target_link_libraries(native_files_c_test PRIVATE platform-native-files)
  set_target_properties(native_files_c_test PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
  add_test(NAME native_files_c COMMAND native_files_c_test)
endif()
foreach(check IN ITEMS platform_services file_services user_services text_services)
  add_executable(${check}_test "../${check}_test.cpp")
  target_link_libraries(${check}_test PRIVATE platform-services)
  add_test(NAME ${check} COMMAND ${check}_test)
endforeach()
foreach(check IN ITEMS settings_store settings_adapter ${settings_native_test})
  add_executable(${check}_test "../${check}_test.cpp")
  target_link_libraries(${check}_test PRIVATE platform-services)
  if(MSVC)
    target_compile_options(${check}_test PRIVATE /UNDEBUG)
  else()
    target_compile_options(${check}_test PRIVATE -UNDEBUG)
  endif()
  add_test(NAME ${check} COMMAND ${check}_test)
endforeach()
add_executable(native_settings_c_test ../native_settings_c_test.c)
target_link_libraries(native_settings_c_test PRIVATE platform-native-settings)
set_target_properties(native_settings_c_test PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE C)
if(MSVC)
  target_compile_options(native_settings_c_test PRIVATE /UNDEBUG)
else()
  target_compile_options(native_settings_c_test PRIVATE -UNDEBUG)
endif()
add_test(NAME native_settings_c COMMAND native_settings_c_test)
add_executable(text_portable_test ../text_services_test.cpp "${platform_dir}/platform_text.cpp" "${platform_dir}/text_utf8.cpp")
target_include_directories(text_portable_test PRIVATE "${platform_dir}")
target_compile_definitions(text_portable_test PRIVATE KAUAI_PORTABLE_TEXT)
target_link_libraries(text_portable_test PRIVATE Threads::Threads platform-engine-text)
add_test(NAME text_portable COMMAND text_portable_test)
