cmake_minimum_required(VERSION 3.22)
project(RetailOracle LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "The retail oracle must be built with a 32-bit Windows compiler")
endif()
if(NOT MSVC)
  message(FATAL_ERROR "The bridge requires an MSVC-compatible compiler for x86 SEH and calling conventions")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(retail-oracle SHARED bridge.cpp)
target_link_libraries(retail-oracle PRIVATE user32 gdi32 advapi32)
if(MSVC)
  target_link_options(retail-oracle PRIVATE "/EXPORT:OracleStart=_OracleStart@4")
endif()
add_executable(retail-oracle-launch launch.cpp)
target_link_libraries(retail-oracle-launch PRIVATE advapi32)

option(RETAIL_ORACLE_INTEGRATION_TESTS "Enable tests which launch retail on the interactive Windows desktop" OFF)
if(RETAIL_ORACLE_INTEGRATION_TESTS)
  enable_testing()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import PIL"
    RESULT_VARIABLE pillow_status OUTPUT_QUIET ERROR_QUIET)
  if(NOT pillow_status EQUAL 0)
    message(FATAL_ERROR "Pillow is required; set Python3_EXECUTABLE to the Python environment containing it")
  endif()
  set(RETAIL_ORACLE_ISO "" CACHE FILEPATH "Original 3D Movie Maker ISO")
  set(RETAIL_ORACLE_SDL "" CACHE FILEPATH "SDL executable built with STUDIO_TEST_TOOLS=ON")
  set(RETAIL_ORACLE_FIXTURE "" CACHE FILEPATH "preview-scene-fixture executable")
  foreach(path RETAIL_ORACLE_ISO RETAIL_ORACLE_SDL RETAIL_ORACLE_FIXTURE)
    if(NOT EXISTS "${${path}}")
      message(FATAL_ERROR "Set ${path} to an existing file")
    endif()
  endforeach()
  get_filename_component(repo "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  add_test(NAME retail-oracle-workflow COMMAND "${Python3_EXECUTABLE}"
    "${repo}/tests/retail_oracle_workflow.py" --sdl "${RETAIL_ORACLE_SDL}" --iso "${RETAIL_ORACLE_ISO}"
    --bridge "$<TARGET_FILE_DIR:retail-oracle>" --fixture "${RETAIL_ORACLE_FIXTURE}"
    --output "${repo}/tmp/retail-oracle-workflow")
  set_tests_properties(retail-oracle-workflow PROPERTIES TIMEOUT 180 ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
endif()
